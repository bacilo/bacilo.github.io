Create `src/pages/publication/` directory and `[...slug].astro` file.

The `[...slug]` (rest parameter) syntax allows matching paths like `/publication/2008-01-01-License-to-chill...`.

```astro
---
import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';

export async function getStaticPaths() {
  const publications = await getCollection('publications');

  return publications.map(pub => {
    // Extract slug from permalink (e.g., "/publication/2008-01-01-Name" -> "2008-01-01-Name")
    const permalink = pub.data.permalink;
    const slug = permalink.replace(/^\/publication\//, '');

    return {
      params: { slug },
      props: { publication: pub },
    };
  });
}

const { publication } = Astro.props;
const { Content } = await publication.render();
```

Create `src/pages/posts/` directory and `[...slug].astro` file.

Posts have a more complex permalink format: `/posts/2012/08/blog-post-1/` (with year/month subdirectories).

```astro
---
import { getCollection } from 'astro:content';
import BaseLayout from '../../layouts/BaseLayout.astro';

export async function getStaticPaths() {
  const posts = await getCollection('posts');

  return posts.map(post => {
    // Extract slug from permalink (e.g., "/posts/2012/08/blog-post-1/" -> "2012/08/blog-post-1")
    // Handle both with and without trailing slash
    const permalink = post.data.permalink || '';
    let slug = permalink.replace(/^\/posts\//, '').replace(/\/$/, '');

    // If no permalink in frontmatter, generate from date + filename
    if (!slug) {
      const date = post.data.date;
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, '0');
      // Extract post name from id (filename without date prefix)
      const postName = post.id.replace(/^\d{4}-\d{2}-\d{2}-/, '').replace(/\.md$/, '');
      slug = `${year}/${month}/${postName}`;
    }

    return {
      params: { slug },
      props: { post },
    };
  });
}

const { post } = Astro.props;
const { Content } = await post.render();

// Format tags for display
const tags = post.data.tags || [];
---

<BaseLayout title={post.data.title}>
  <article class="post">
    <header>
      <h1>{post.data.title}</h1>
      <p class="meta">
        <time datetime={post.data.date.toISOString()}>
          {post.data.date.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' })}
        </time>
      </p>
      {tags.length > 0 && (
        <div class="tags">
          {tags.map(tag => (
            <span class="tag">{tag}</span>
          ))}
        </div>
      )}
    </header>

    <section class="content">
      <Content />
    </section>
  </article>
</BaseLayout>
```
